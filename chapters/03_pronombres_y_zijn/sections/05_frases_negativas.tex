\section{Frases negativas con \textit{zijn}}
\textbf{Estructura:} Sujeto + \textit{zijn} + \textit{niet} + complemento/atributo.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik ben niet moe.}
  \item \textit{Hij is niet in Brussel.}
  \item \textit{We zijn niet Nederlands.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Convierte 6 afirmaciones en negativas colocando \textit{niet} al final.
  \item Di y escribe las frases, cuidando el orden V2 en oraciones con adverbio inicial (\textit{Vandaag ben ik niet thuis}).
\end{itemize}
