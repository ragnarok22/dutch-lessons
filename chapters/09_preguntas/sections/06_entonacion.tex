\section{Entonación en preguntas}
\textbf{Reglas rápidas}
\begin{itemize}[leftmargin=*]
  \item Sí/no: subida ligera al final (especialmente en informal).
  \item Wh- preguntas: entonación más estable/descendente.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Graba 6 preguntas sí/no y 6 wh-, compara curvas de entonación.
  \item Haz shadowing de un diálogo corto centrado en preguntas.
\end{itemize}
