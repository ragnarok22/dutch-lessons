\section{Sustantivos contables básicos}
\textbf{Qué es:} Palabras de uso frecuente que conviene practicar en singular/plural desde el inicio.

\textbf{Mini lista}
\begin{itemize}[leftmargin=*]
  \item Objetos: \textit{boek/boeken, tafel/tafels, stoel/stoelen, tas/tassen}.
  \item Personas: \textit{vriend/vrienden, kind/kinderen, docent/docenten}.
  \item Comida: \textit{appel/appels, brood/broden, ei/eieren}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Haz pares de tarjetas singular/plural y barájalas a diario.
  \item Crea 8 frases mezclando contables de cada grupo.
\end{itemize}
