\section{Frases con adverbio inicial}
\textbf{Patrón:} Adverbio/expresión inicial + verbo + sujeto + resto.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Morgen ga ik werken.}
  \item \textit{Daar woont hij.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Reescribe 8 frases moviendo tiempo/lugar al inicio y manteniendo V2.
  \item Lee en voz alta para interiorizar la inversión sujeto-verbo.
\end{itemize}
