\section{Plural en \textit{-en}}
\textbf{Regla general:} La mayoría forma plural añadiendo \textit{-en}; posibles ajustes de consonantes/vocales.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{boek \textrightarrow{} boeken}
  \item \textit{huis \textrightarrow{} huizen} (cambio vocálico ui- en plural)
  \item \textit{kat \textrightarrow{} katten} (doble consonante para mantener vocal corta)
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Convierte 12 singulares a plural con \textit{-en}, cuidando duplicar consonante si la vocal debe seguir corta.
  \item Lee en voz alta contrastando vocal corta/larga antes y después del cambio.
\end{itemize}
