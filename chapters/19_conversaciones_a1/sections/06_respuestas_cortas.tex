\section{Respuestas cortas y naturales}
\textbf{Afirmar y negar rápidamente.}
\begin{itemize}[leftmargin=*]
  \item Sí/no + verbo: \textit{Ja, dat is zo}.~\textit{Nee, dat is niet zo}. 
  \item Repetir el verbo para sonar natural: \textit{Ja, dat klopt}.~\textit{Nee, ik kan niet}. 
  \item Cortesía: \textit{Ja graag}, \textit{Nee, dank je}. 
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 6 respuestas cortas (3 afirmaciones, 3 negaciones) a preguntas comunes.
  \item Practica transformarlas en audio con buena entonación.
\end{itemize}
