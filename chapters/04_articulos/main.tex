\chapter{Artículos \textit{de} / \textit{het} / \textit{een}}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Diferenciar sustantivos de-woorden y het-woorden en vocabulario frecuente A1.
	\item Usar artículos definidos e indefinidos en singular y plural.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Artículos: \textit{de} (común), \textit{het} (neutro), \textit{een} (indefinido, sin énfasis vocálico).
	\item Patrones frecuentes de \textit{het}: diminutivos en \textit{-je}, idiomas, metales; regla práctica.
	\item Plural con artículo: \textit{de boeken, de huizen}.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Clasificar vocabulario nuevo en lista de-woorden/het-woorden.
	\item Completar oraciones con el artículo correcto.
	\item Juego de memoria con tarjetas por género.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item 20 oraciones de elección múltiple de/het + traducción inversa.
\end{itemize}
