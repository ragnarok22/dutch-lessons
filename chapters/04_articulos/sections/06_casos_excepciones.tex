\section{Casos comunes y excepciones}
\textbf{Objetivo:} Reconocer patrones y palabras que suelen causar duda.

\textbf{Hacks rápidos}
\begin{itemize}[leftmargin=*]
  \item Diminutivos siempre \textit{het}: \textit{het huisje, het kopje}.
  \item Palabras compuestas heredan el artículo del núcleo final: \textit{de sleutel + het gat = het sleutelgat}; \textit{de stad + het centrum = het stadscentrum}.
  \item Profesiones suelen ser \textit{de}: \textit{de dokter, de leraar}.
\end{itemize}

\textbf{Lista corta a memorizar (het)}
\begin{itemize}[leftmargin=*]
  \item \textit{het kind, het ei, het water, het geld, het weer}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Crea 10 tarjetas con estas excepciones y repasa a diario.
  \item Haz frases con compuestos para fijar el artículo del núcleo.
\end{itemize}
