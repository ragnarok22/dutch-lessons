\section{Verbos comunes de uso diario}
\textbf{Lista base}
\begin{itemize}[leftmargin=*]
  \item \textit{werken} (trabajar), \textit{wonen} (vivir), \textit{leren} (aprender), \textit{spelen} (jugar), \textit{maken} (hacer), \textit{koken} (cocinar), \textit{luisteren} (escuchar), \textit{kijken} (mirar).
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Elige 4 verbos y crea un párrafo de 80 palabras en presente.
  \item Haz tarjetas verbo + 2 frases modelo (una afirmativa, una negativa).
\end{itemize}
