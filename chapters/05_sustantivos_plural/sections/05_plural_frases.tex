\section{Uso del plural en frases}
\textbf{Regla rápida con números:} Singular para 1, plural para 2+ y sin artículo: \textit{twee boeken, drie huizen}.

\textbf{Con determinantes}
\begin{itemize}[leftmargin=*]
  \item \textit{veel/weinig} + plural: \textit{veel mensen, weinig auto's}.
  \item Demostrativos plurales: \textit{deze, die} + plural: \textit{deze kinderen}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Completa 10 frases con número + sustantivo correcto.
  \item Reescribe 5 frases de singular a plural ajustando artículo/determinante.
\end{itemize}
