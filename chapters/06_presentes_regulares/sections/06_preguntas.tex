\section{Preguntas con verbos regulares}
\textbf{Sí/no:} Verbo conjugado en primera posición.
\begin{itemize}[leftmargin=*]
  \item \textit{Werk je morgen?} \textit{Speelt hij gitaar?}
\end{itemize}

\textbf{Wh- preguntas:} Palabra interrogativa + verbo + sujeto.
\begin{itemize}[leftmargin=*]
  \item \textit{Waar woon je?} \textit{Wat maak jij?}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 6 preguntas (3 sí/no, 3 wh-) con verbos regulares del capítulo.
  \item Role-play: entrevista breve sobre rutina y hobbies.
\end{itemize}
