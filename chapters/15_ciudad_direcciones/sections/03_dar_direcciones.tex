\section{Dar direcciones simples}
\textbf{Imperativos y marcadores.}
\begin{itemize}[leftmargin=*]
  \item Imperativos: \textit{Ga rechtdoor}, \textit{Neem de eerste straat links/rechts}, \textit{Steek de straat over}.
  \item Distancias: \textit{na 200 meter}, \textit{om de hoek} (a la vuelta), \textit{vlakbij} (cerca), \textit{ver} (lejos).
  \item Puntos de referencia: \textit{bij de kerk}, \textit{naast de supermarkt}, \textit{tegenover het station}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Da tres instrucciones para llegar desde la estación a la plaza principal.
  \item Escribe un mini diálogo de 4 líneas dando indicaciones.
\end{itemize}
