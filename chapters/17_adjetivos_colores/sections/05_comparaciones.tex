\section{Comparaciones simples}
\textbf{Formación.}
\begin{itemize}[leftmargin=*]
  \item Grado comparativo: +\textit{er}. Ej.: \textit{groot} → \textit{groter}, \textit{klein} → \textit{kleiner}.
  \item Superlativo: +\textit{st} con artículo \textit{het} + \textit{-e} si va atributivo: \textit{het grootste huis}.
  \item Irregulares frecuentes: \textit{goed → beter → het beste}, \textit{slecht → slechter → het slechtste}.
\end{itemize}

\textbf{Estructuras}
\begin{itemize}[leftmargin=*]
  \item \textit{X is groter dan Y} (más grande que)
  \item \textit{Dit is het goedkoopste product}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Compara dos objetos de tu mesa con \textit{groter/kleiner}.
  \item Escribe dos frases con superlativo usando \textit{het}.
\end{itemize}
