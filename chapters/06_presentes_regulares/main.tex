\chapter{Verbos en presente (regulares)}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Conjugar verbos regulares en presente de indicativo.
	\item Usar patrones de terminación y reconocer variaciones ortográficas menores.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item De infinitivo a raíz: quitar \textit{-en} y ajustar ortografía.
	\item Terminaciones por persona y omisión de \textit{-t} en \textit{jij} inversión.
	\item Orden de la oración en afirmaciones, negaciones y preguntas con verbos regulares.
	\item Lista base de verbos A1 de alta frecuencia.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Verbos: \textit{werken}, \textit{wonen}, \textit{leren}, \textit{spreken}, \textit{kijken}, \textit{koken}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item No pronuncies la \textit{-en} final: \textit{werken} → /ˈʋɛrkən/.
	\item Marca la \textit{t} en formas con \textit{-t} antes de vocal: \textit{jij werkt} → /ʋɛrkt/.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Olvidar quitar \textit{-t} en inversión con \textit{jij}: \textit{Werk jij\ldots?}
	\item Repetir sujeto al leer: evita \textit{Ik ik werk}.
	\item Colocar \textit{niet} justo después del verbo; en V2 va hacia el final.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Conjuga 10 verbos en todas las personas con autocorrección.
	\item Transforma 6 afirmaciones en negaciones y luego en preguntas.
	\item Ordena palabras desordenadas para reconstruir V2 correcto.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe 10 frases sobre tu día usando diferentes sujetos y verbos.
	\item Graba 8 preguntas cortas a un compañero usando inversión con \textit{jij/jullie}.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Formas la raíz y terminaciones sin dudar.
	\item Usas V2 en afirmaciones y preguntas con regulares.
	\item Colocas \textit{niet} en lugar correcto en nivel A1.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes conjugar verbos regulares y construir frases/preguntas básicas sobre rutinas con orden correcto y buena pronunciación.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Escribe 12 frases usando 6 verbos regulares en distintas personas.
\end{itemize}
