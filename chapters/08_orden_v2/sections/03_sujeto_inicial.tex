\section{Frases con sujeto inicial}
\textbf{Patrón:} Sujeto + verbo + complementos (orden simple).

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik woon in Utrecht.}
  \item \textit{Zij werkt in een ziekenhuis.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 6 frases con sujeto inicial usando distintos verbos regulares e irregulares.
  \item Añade un adverbio al final y verifica que no rompe V2.
\end{itemize}
