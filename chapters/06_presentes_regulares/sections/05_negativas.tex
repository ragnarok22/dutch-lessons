\section{Frases negativas}
\textbf{Estructura básica:} Sujeto + verbo (V2) + resto + \textit{niet}.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik werk vandaag niet.}
  \item \textit{Hij speelt niet buiten.}
  \item \textit{We leren niet in de avond.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Convierte 6 afirmaciones en negativas, colocando \textit{niet} al final.
  \item Añade un adverbio inicial y verifica el orden: \textit{Morgen werk ik niet}.
\end{itemize}
