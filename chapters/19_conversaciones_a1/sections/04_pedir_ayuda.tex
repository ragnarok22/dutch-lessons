\section{Pedir ayuda}
\textbf{Frases clave.}
\begin{itemize}[leftmargin=*]
  \item \textit{Kunt u mij helpen?} / \textit{Kan je me helpen?}
  \item \textit{Ik begrijp het niet}. (No lo entiendo.)
  \item \textit{Kunt u dat herhalen, alstublieft?}
  \item \textit{Waar is de wc?} / \textit{Waar is de uitgang?}
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Role-play pidiendo ayuda en la calle y recibiendo una dirección corta.
  \item Escribe dos preguntas de aclaración para clase o tienda.
\end{itemize}
