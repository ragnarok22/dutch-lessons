\chapter{Orden de la frase (V2)}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Aplicar la regla de verbo en segunda posición en enunciados afirmativos.
	\item Mover el foco inicial (tiempo/lugar) manteniendo V2.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Estructura básica S-V2-resto y variantes con adverbio inicial.
	\item Combinación con negación y complementos cortos.
	\item Patrón de orden en frases con verbo separable (vista previa).
	\item Detección y corrección de errores de posición.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Adverbios iniciales típicos: \textit{vandaag}, \textit{morgen}, \textit{nu}, \textit{thuis}, \textit{buiten}, \textit{daarna}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Marca pausa breve tras el primer elemento cuando no es el sujeto para resaltar V2.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Dejar verbo en tercera posición tras adverbio inicial.
	\item Colocar dos elementos antes del verbo (p.ej., tiempo + lugar) sin coma ni ajuste.
	\item Olvidar mover el sujeto tras el verbo cuando inicia un adverbio.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Reordena 12 frases mezclando tiempo/lugar inicial para cumplir V2.
	\item Completa huecos con adverbio inicial dado.
	\item Detecta y corrige errores de V2 en ejemplos cortos.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe 10 frases con adverbio inicial variado; graba 5.
	\item Redacta un mini párrafo de 6 líneas sobre tu día usando al menos 4 V2 distintos.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Identificas el verbo principal y lo colocas en segunda posición.
	\item Sabes mover complementos iniciales sin perder V2.
	\item Detectas y corriges errores frecuentes de orden.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes producir frases con V2 correcto cambiando el foco inicial y sonar natural en narraciones simples.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item 12 frases: 6 con sujeto inicial, 6 con tiempo/lugar inicial, todas con V2 correcto.
\end{itemize}
