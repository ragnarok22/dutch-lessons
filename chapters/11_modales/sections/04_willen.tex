\section{\textit{willen} (querer, deseo/intención)}
\textbf{Significado.} Expresa intención o deseo; puede sonar directo. Suaviza con \textit{graag} para cortesía.

\begin{itemize}[leftmargin=*]
  \item Formas: \textit{ik wil}, \textit{jij wil}/\textit{jij wilt}, \textit{hij/zij wil}, \textit{wij/jullie/zij willen}.
  \item Fórmula cortés en compras/pedidos: \textit{Ik wil graag koffie}.
\end{itemize}

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik wil Nederlands leren}. (Quiero aprender neerlandés.)
  \item \textit{Wil je mee?} (¿Quieres acompañar?)
  \item \textit{Wij willen niet wachten}. (No queremos esperar.)
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Pide algo en un café usando \textit{wil graag}.
  \item Haz una pregunta informal con \textit{Wil je \ldots?}
\end{itemize}
