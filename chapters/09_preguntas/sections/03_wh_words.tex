\section{Palabras interrogativas (wie, wat, waar, wanneer, waarom, hoe)}
\textbf{Patrón:} Wh-word + verbo + sujeto + resto.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Wie is dat?}
  \item \textit{Waar werk je?}
  \item \textit{Wanneer kom je?}
  \item \textit{Waarom ga je?}
  \item \textit{Hoe gaat het?}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Crea 10 preguntas, una con cada wh-word, mezclando verbos regulares e irregulares.
  \item Role-play: entrevista corta usando al menos 4 wh-words.
\end{itemize}
