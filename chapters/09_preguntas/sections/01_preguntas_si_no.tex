\section{Preguntas sí/no}
\textbf{Regla:} Verbo conjugado en primera posición.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ben je klaar?}
  \item \textit{Werk jij morgen?}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Transforma 8 enunciados en preguntas sí/no.
  \item Añade partícula de tiempo/lugar al final y lee en voz alta.
\end{itemize}
