\chapter{Pronombres personales y verbo \textit{zijn}}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Memorizar los pronombres sujeto y las formas del verbo \textit{zijn} en presente.
	\item Construir oraciones básicas afirmativas, interrogativas y negativas con \textit{zijn}.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Pronombres: \textit{ik, je/jij, u, hij, zij/ze, het, we/wij, jullie, ze}.
	\item Conjugación de \textit{zijn}: \textit{ben, bent, is, zijn}.
	\item Contracciones frecuentes: \textit{\textquotesingle{}k ben, \textquotesingle{}t is} (exposición).
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Tablas de sustitución con profesiones, nacionalidades y estados de ánimo.
	\item Preguntas-respuestas rápidas: \textit{Ben je \ldots{}? Ja, ik ben \ldots{} / Nee, ik ben niet \ldots{}}.
	\item Mini dictados de frases.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Escribe y lee en voz alta 10 frases usando todas las personas gramaticales.
\end{itemize}
