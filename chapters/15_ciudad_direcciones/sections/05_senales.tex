\textbf{Palabras frecuentes.}
\begin{itemize}[leftmargin=*]
  \item Tráfico: \textit{uitgang} (salida), \textit{ingang} (entrada), \textit{omleiding} (desvío), \textit{verboden toegang} (prohibido pasar), \textit{parkeren} (aparcar).
  \item Servicio público: \textit{toilet}, \textit{lift}, \textit{balie} (mostrador), \textit{wachtruimte} (sala de espera).
\end{itemize}
