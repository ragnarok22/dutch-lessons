\section{Hablar de horarios}
\textbf{Expresiones de tiempo.}
\begin{itemize}[leftmargin=*]
  \item Horas: \textit{om acht uur}, \textit{om half negen}. 
  \item Frecuencia: \textit{altijd, meestal, soms, nooit, elke dag, in het weekend}.
  \item Secuencia: \textit{eerst}, \textit{dan}, \textit{daarna}, \textit{tot}, \textit{na}.
\end{itemize}

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik begin om negen uur en ik stop om vijf uur}. 
  \item \textit{Daarna eet ik lunch}. 
  \item \textit{Ik werk meestal tot zes uur}. 
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Redacta tu horario de lunes a viernes con horas precisas.
  \item Ordena tres frases usando conectores de secuencia.
\end{itemize}
