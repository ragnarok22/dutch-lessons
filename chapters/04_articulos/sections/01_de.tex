\section{Sustantivos con \textit{de}}
\textbf{Qué es:} Artículo definido para género común (de-woorden). Abarca la mayoría de sustantivos.

\textbf{Patrones frecuentes}
\begin{itemize}[leftmargin=*]
  \item Personas, profesiones, nacionalidades: \textit{de leraar, de arts, de Spanjaard}.
  \item Plurales siempre usan \textit{de}: \textit{de boeken, de huizen}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Clasifica 15 palabras nuevas en lista de-woorden.
  \item Completa oraciones: \textit{\ldots{} man is in Amsterdam}; \textit{\ldots{} kinderen spelen buiten}.
\end{itemize}
