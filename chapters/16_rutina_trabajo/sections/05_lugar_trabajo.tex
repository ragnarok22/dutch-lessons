\section{Lugar de trabajo}
\textbf{Describir el sitio.}
\begin{itemize}[leftmargin=*]
  \item Tipo de lugar: \textit{op kantoor, in een winkel, in een fabriek, thuis, buiten}.
  \item Herramientas/espacios: \textit{bureau} (escritorio), \textit{vergaderruimte} (sala de reuniones), \textit{keuken}, \textit{werkplaats}.
\end{itemize}

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik werk thuis twee dagen per week}. 
  \item \textit{We hebben een kleine vergaderruimte}. 
  \item \textit{Hij werkt buiten op een bouwplaats}. 
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Describe tu lugar de trabajo en 3 frases.
  \item Menciona dos herramientas o espacios que usas a diario.
\end{itemize}
