\chapter{Pronunciación y alfabeto}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Reconocer el alfabeto neerlandés y sus sonidos principales.
	\item Producir vocales largas/cortas y diptongos con buena aproximación.
	\item Diferenciar consonantes problemáticas: \textit{g/ch}, \textit{r}, \textit{ui}, \textit{ij/ei}.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Alfabeto y nombres de letras.
	\item Vocales simples vs.\ dobles; diptongos \textit{ui, ij/ei, ou/au, eu}.
	\item Consonantes fricativas \textit{g/ch}, sonora \textit{v/z}, suave \textit{r}.
	\item Reglas básicas de acento y ritmo silábico.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Escucha y repetición de pares mínimos.
	\item Dictado de letras y deletreo de tu nombre.
	\item Shadowing de frases cortas centradas en sonidos nuevos.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Graba un minuto presentando el alfabeto y tu nombre; compáralo con un modelo nativo.
\end{itemize}
