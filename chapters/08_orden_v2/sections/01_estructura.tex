\section{Estructura básica de la oración}
\textbf{Esqueleto V2:} [Elemento inicial] + Verbo conjugado + [Sujeto si no va primero] + Complementos.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item Sujeto inicial: \textit{Ik lees een boek.}
  \item Tiempo inicial: \textit{Morgen lees ik een boek.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Identifica el verbo y colócalo siempre en la segunda posición en 10 frases mezcladas.
  \item Reescribe frases moviendo un elemento inicial y ajustando el orden.
\end{itemize}
