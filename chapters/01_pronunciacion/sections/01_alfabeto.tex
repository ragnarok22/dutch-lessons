\section{Alfabeto holandés}
\textbf{Qué es y para qué sirve:} Inventario de letras y sus nombres. Dominarlo permite deletrear, buscar en diccionarios y entender deletreos en audio.

\textbf{Cómo aplicarlo:}
\begin{itemize}[leftmargin=*]
  \item Aprende los nombres de las letras: \textit{a, bee, cee, dee, ee, ef, gee, haa, ie, jee, kaa, el, em, en, oo, pee, kuu, er, es, tee, uu, vee, w, iks, ij, zet}.
  \item Practica deletrear tu nombre, correo y ciudad.
  \item Usa dictados cortos de letras para afinar contraste \textit{e/ee}, \textit{i/ie}, \textit{u/uu}.
\end{itemize}

\textbf{Ejemplo breve:}
\enquote{Hoe spel je dat?} \enquote{\textit{R-O-T-T-E-R-D-A-M}.}
