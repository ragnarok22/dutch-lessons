\section{Descripción de objetos y personas}
\textbf{Plantillas rápidas.}
\begin{itemize}[leftmargin=*]
  \item Objeto: \textit{Het is een \textit{kleur} + \textit{adjetivo} + \textit{sustantivo}}. Ej.: \textit{Het is een blauwe kleine tas}.
  \item Persona: \textit{Hij/Zij is + adjetivo}; \textit{Hij/Zij heeft + adjetivo + haar/ogen}.
\end{itemize}

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Zij heeft lang blond haar}.
  \item \textit{Dit is een grote bruine tafel}.
  \item \textit{Mijn vriend is heel grappig}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Describe dos objetos cercanos con color y tamaño.
  \item Escribe dos frases sobre personas usando adjetivos de carácter y apariencia.
\end{itemize}
