\chapter{Comida y bebida}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Nombrar alimentos y bebidas comunes y hacer pedidos simples.
	\item Practicar cantidades básicas y preferencias.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Vocabulario: comidas del día, supermercado, café/restaurante.
	\item Estructuras: \textit{Ik wil graag \ldots{}}, \textit{Mag ik \ldots{}?}, \textit{Ik houd van \ldots{} / Ik lust geen \ldots{}}.
	\item Cantidades: \textit{een glas, een kop, een broodje, kilo, liter}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Alarga vocal en \textit{kaas}, \textit{brood}; evita acortar.
	\item Pronuncia /x/ suave en \textit{graag} al pedir: /xraːx/.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Usar \textit{hebben} en lugar de \textit{willen} al pedir: di \textit{Ik wil graag\ldots{}}.
	\item Olvidar plural de conteo: \textit{twee broodjes}, no \textit{twee broodje}.
	\item Mezclar \textit{houden van} (gusto general) con \textit{lusten} (comida/bebida).
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Rellena menús con precios y haz pedidos guiados.
	\item Completa frases con cantidad correcta (\textit{een glas}, \textit{een kilo}, \textit{twee stukken}).
	\item Transforma gustos afirmativos en negativos con \textit{geen/niet}.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe y graba un pedido completo en café y en supermercado (8–10 ítems).
	\item Redacta tus preferencias de comida en 8 frases (\textit{Ik houd van\ldots{}}, \textit{Ik lust geen\ldots{}}).
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Pides comida y bebida con cortesía usando \textit{wil graag} o \textit{mag ik}.
	\item Manejas cantidades básicas sin traducir del español.
	\item Distingues \textit{houden van} vs. \textit{lusten} para gustos.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes pedir en cafetería/restaurante y hablar de lo que te gusta o no te gusta con vocabulario de comida A1.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Simulación de pedir en cafetería y leer un menú simple.
	\item Lista de compras con precios estimados.
	\item Recetas cortas con imperativos básicos (exposición).
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Role-play de pedido en 6 líneas; lista de 10 ítems de compra.
\end{itemize}
