\chapter{Preguntas y palabras interrogativas}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Formular preguntas de sí/no y de información respetando la posición del verbo.
	\item Manejar palabras interrogativas básicas.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Preguntas sí/no: verbo en primera posición.
	\item Wh-words: \textit{wie, wat, waar, wanneer, waarom, hoe, hoeveel, welke}.
	\item Entonación ascendente en preguntas informales.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Transformar enunciados en preguntas de sí/no y de información.
	\item Role-play con tarjetas de pregunta-respuesta.
	\item Juego de encuesta breve entre compañeros.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item 10 preguntas orales grabadas + respuestas cortas.
\end{itemize}
