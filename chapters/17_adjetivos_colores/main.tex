\chapter{Adjetivos y colores}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Usar adjetivos básicos para describir objetos y personas.
	\item Aplicar reglas simples de posición y adjetivación con \textit{de/het} y diminutivos.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Colores comunes y adjetivos de tamaño, forma y estado.
	\item Adjetivos atributivos vs.\ predicativos; \textit{-e} en adjetivos delante de sustantivo con \textit{de} y la mayoría de \textit{het} (excepción con indefinido neutro).
	\item Orden: color + material + sustantivo (introducción).
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Alarga vocal en colores como \textit{blauw} \textipa{/blAu/} y \textit{groen} \textipa{/xrUn/}; evita españolizar.
	\item Suaviza \textit{g} en \textit{geel}, \textit{grijs}; no /g/ dura.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Omitir \textit{-e} en adjetivos ante \textit{de}-woorden: \textit{de grote tafel}.
	\item Añadir \textit{-e} tras \textit{een} + \textit{het}-woord cuando es indefinido: \textit{een klein huis} (sin -e).
	\item Orden incorrecto: color va después de determinante y antes del sustantivo.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Completa frases seleccionando forma con/ sin \textit{-e}.
	\item Reescribe descripciones moviendo color y tamaño al lugar correcto.
	\item Pareo de fotos con adjetivos proporcionados.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Describe 8 objetos cercanos con dos adjetivos cada uno.
	\item Graba 6 frases comparando cosas (\textit{kleiner dan}, \textit{mooier dan}).
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Añades o quitas \textit{-e} según artículo y género.
	\item Colocas color y tamaño en orden natural.
	\item Formas comparativos simples con claridad.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes describir objetos y personas con colores y adjetivos básicos, usando concordancia y orden correctos.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Describir objetos en el aula o casa.
	\item Juego de ``encuentra el objeto'' con pistas adjetivales.
	\item Parear fotos con descripciones.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item 10 descripciones breves (2 frases cada una) de objetos o personas.
\end{itemize}
