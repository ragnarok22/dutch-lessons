\section{Colores básicos}
\textbf{Lista.} \textit{rood, oranje, geel, groen, blauw, paars, roze, zwart, wit, grijs, bruin}.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{De jas is rood}. 
  \item \textit{Het huis is blauw}. 
  \item \textit{Ik draag graag zwarte schoenen}. 
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Nombra 5 objetos cercanos con color en neerlandés.
  \item Haz tarjetas (color + objeto) y léelas en voz alta.
\end{itemize}
