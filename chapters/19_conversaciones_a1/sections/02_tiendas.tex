\section{Conversaciones en tiendas}
\textbf{Frases útiles.}
\begin{itemize}[leftmargin=*]
  \item Saludo y oferta de ayuda: \textit{Hallo, kan ik u helpen?}
  \item Pedir un producto: \textit{Ik zoek een jas / Ik neem deze}.
  \item Tallas y precios: \textit{Welke maat? Hoeveel kost het?}
  \item Pago: \textit{Kan ik pinnen? Mag ik de bon?}
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Role-play cliente/vendedor con dos productos y un precio.
  \item Escribe dos preguntas sobre talla y color.
\end{itemize}
