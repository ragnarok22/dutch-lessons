\section{Respuestas cortas}
\textbf{Objetivo:} Responder de forma natural con verbo auxiliar/lexical + pronombre.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Werk je hier? Ja, ik werk hier. / Nee, ik werk hier niet.}
  \item \textit{Ga je mee? Ja, ik ga mee. / Nee, ik ga niet mee.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe dos respuestas (sí/no) para 8 preguntas del capítulo.
  \item Lee en voz alta cuidando entonación descendente en respuestas.
\end{itemize}
