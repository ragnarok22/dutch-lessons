\section{Plural en \textit{-s}}
\textbf{Cuándo usarlo:} Sustantivos que terminan en vocal no acentuada o en \textit{-el, -em, -en, -er} suelen tomar \textit{-s}.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item Vocal final: \textit{auto \textrightarrow{} auto's}, \textit{cafe \textrightarrow{} cafe's}.
  \item Final en \textit{-el/-er}: \textit{tafel \textrightarrow{} tafels}, \textit{computer \textrightarrow{} computers}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Lista 10 sustantivos y decide \textit{-s} o \textit{-en} (mezcla de finales).
  \item Lee en voz alta cuidando la \textit{s} final suave.
\end{itemize}
