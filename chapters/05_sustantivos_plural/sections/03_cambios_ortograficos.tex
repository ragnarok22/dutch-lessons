\section{Cambios ortográficos en el plural}
\textbf{Qué son:} Ajustes para mantener vocal corta/larga o evitar combinaciones raras.

\textbf{Casos típicos}
\begin{itemize}[leftmargin=*]
  \item Duplicar consonante para vocal corta: \textit{kat \textrightarrow{} katten}, \textit{bed \textrightarrow{} bedden}.
  \item Quitar una vocal doble para vocal larga en sílaba abierta: \textit{baan \textrightarrow{} banen}.
  \item Cambio de \textit{-f} a \textit{-v} y \textit{-s} a \textit{-z}: \textit{dief \textrightarrow{} dieven}, \textit{huis \textrightarrow{} huizen}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 8 palabras con cambio f/v o s/z y sus plurales.
  \item Divide en sílabas antes y después para ver la razón del cambio.
\end{itemize}
