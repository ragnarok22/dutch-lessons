\section{Presentaciones completas}
\textbf{Estructura breve.}
\begin{itemize}[leftmargin=*]
  \item Nombre y origen: \textit{Ik ben Ana. Ik kom uit Spanje}.
  \item Residencia y lengua: \textit{Ik woon in Utrecht. Ik spreek Spaans en een beetje Nederlands}.
  \item Ocupación: \textit{Ik ben student / ik werk als verpleegkundige}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Graba o escribe una presentación de 4 líneas.
  \item Intercambia presentaciones con un compañero y responde una pregunta.
\end{itemize}
