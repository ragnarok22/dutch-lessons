\section{Uso del verbo \textit{zijn}}
\textbf{Qué es:} Verbo copulativo "ser/estar" para identidad, estado y localización general.

\textbf{Cuándo se usa}
\begin{itemize}[leftmargin=*]
  \item Identidad y descripción: \textit{Ik ben student.}
  \item Origen y ubicación amplia: \textit{Hij is uit Spanje. We zijn in Amsterdam.}
  \item Estado/condición: \textit{Ik ben moe. Het is koud.}
\end{itemize}

\textbf{No se usa} con posesión: se emplea \textit{hebben} (\textit{Ik heb een boek}).
