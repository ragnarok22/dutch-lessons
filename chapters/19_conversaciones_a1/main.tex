\chapter{Conversaciones cotidianas A1}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Integrar vocabulario y gramática previos en diálogos comunes.
	\item Mejorar fluidez en interacciones breves en tiendas, transporte y socialización.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Situaciones: comprar billete, hacer un pedido, pedir ayuda, charlas pequeñas.
	\item Estrategias de reparación: \textit{Sorry? Kunt u dat herhalen? Wat betekent \ldots{}?}
	\item Fórmulas de cortesía y cierre de conversación.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Practica ritmo rápido pero claro en frases cortas de servicio (\textit{Mag ik ... ?}).
	\item Suaviza \textit{g} en \textit{graag}; evita /grag/.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Olvidar cierres corteses: \textit{Dank u wel}, \textit{Fijne dag}.
	\item No usar entonación ascendente en preguntas de servicio.
	\item Traducciones literales largas; mantén frases cortas y directas.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Rellena huecos en diálogos modelo con fórmulas de cortesía.
	\item Repite y graba diálogos cambiando un dato (precio, destino, hora).
	\item Tarjetas de reparación: responde con \textit{Sorry? Kunt u dat herhalen?}
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe 3 diálogos originales de 8 líneas (tienda, restaurante, ayuda en calle).
	\item Graba versiones orales de los diálogos con variaciones de datos.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Lanzas y cierras interacciones cortas con cortesía.
	\item Usas preguntas breves con entonación adecuada.
	\item Reparas comunicación con frases simples cuando no entiendes.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes manejar diálogos cotidianos en tienda, restaurante y calle, pidiendo aclaraciones y cerrando con cortesía.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Dramatizaciones con guiones y luego sin guion.
	\item Grabarse en audio simulando cada situación.
	\item Escucha de diálogos modelo y transcripción parcial.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Tres diálogos de 8 líneas cada uno cubriendo situaciones distintas.
\end{itemize}
