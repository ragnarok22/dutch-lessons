\chapter{Verbos en presente (regulares)}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Conjugar verbos regulares en presente de indicativo.
	\item Usar patrones de terminación y reconocer variaciones ortográficas menores.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Terminaciones: raíz + \textit{-en} (inf), \textit{-t} (2ª/3ª sg), \textit{-en} (pl), forma nula en 1ª sg.
	\item Eliminación de \textit{-en} y ajuste de consonantes dobles/vocales largas.
	\item Verbos modelo: \textit{werken, wonen, leren, maken, spelen}.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Tablas de conjugación y frases guiadas.
	\item Rondas de preguntas/respuestas usando verbos del día.
	\item Mini historias en presente.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Escribe 12 frases usando 6 verbos regulares en distintas personas.
\end{itemize}
