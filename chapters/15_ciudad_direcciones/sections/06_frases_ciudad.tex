\section{Frases útiles en la ciudad}
\textbf{Supervivencia A1.}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik ben verdwaald}. (Estoy perdido.)
  \item \textit{Kunt u mij helpen?} / \textit{Kan je me helpen?}
  \item \textit{Is het hier dichtbij?} / \textit{Hoe ver is het?}
  \item \textit{Dank u wel, fijne dag}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Role-play pidiendo ayuda y agradeciendo.
  \item Escribe dos frases para confirmar la dirección recibida.
\end{itemize}
