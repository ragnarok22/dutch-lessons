\section{Frases afirmativas}
\textbf{Estructura:} [Elemento inicial] + verbo (2ª posición) + sujeto (si no va primero) + resto.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik leer Nederlands.}
  \item \textit{Vandaag werk ik thuis.}
  \item \textit{We spelen in het park.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 8 afirmaciones con distintos verbos y adverbios iniciales (\textit{vandaag, morgen, nu, vaak}).
  \item Lee en voz alta manteniendo V2 correcto.
\end{itemize}
