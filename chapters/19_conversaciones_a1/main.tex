\chapter{Conversaciones cotidianas A1}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
  \item Integrar vocabulario y gramática previos en diálogos comunes.
  \item Mejorar fluidez en interacciones breves en tiendas, transporte y socialización.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
  \item Situaciones: comprar billete, hacer un pedido, pedir ayuda, charlas pequeñas.
  \item Estrategias de reparación: \textit{Sorry? Kunt u dat herhalen? Wat betekent ...?}
  \item Fórmulas de cortesía y cierre de conversación.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
  \item Dramatizaciones con guiones y luego sin guion.
  \item Grabarse en audio simulando cada situación.
  \item Escucha de diálogos modelo y transcripción parcial.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
  \item Tres diálogos de 8 líneas cada uno cubriendo situaciones distintas.
\end{itemize}
