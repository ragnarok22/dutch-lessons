\chapter{Artículos \textit{de} / \textit{het} / \textit{een}}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Diferenciar sustantivos de-woorden y het-woorden en vocabulario frecuente A1.
	\item Usar artículos definidos e indefinidos en singular y plural.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Distribución básica de \textit{de} vs. \textit{het} y reglas prácticas.
	\item Uso de \textit{een} en afirmaciones, negaciones y preguntas.
	\item Acuerdos en plural y con adjetivos básicos.
	\item Casos frecuentes que generan duda y excepciones comunes.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Sustantivos con \textit{de}: \textit{tafel}, \textit{auto}, \textit{straat}, \textit{winkel}, \textit{vrouw}.
	\item Sustantivos con \textit{het}: \textit{huis}, \textit{kind}, \textit{boek}, \textit{water}, \textit{meisje}.
	\item Plurales de referencia: \textit{de huizen}, \textit{de kinderen}, \textit{de boeken}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item La \textit{het} se reduce a \textipa{/@t/} en habla rápida; escucha y repite para sonar natural.
	\item Mantén \textipa{/@/} corta en \textit{een}; evita alargarla como \textit{een} [e:n].
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Generalizar \textit{de} a todo; memoriza los het-woorden más frecuentes.
	\item Olvidar que el plural siempre lleva \textit{de}, incluso si el singular es \textit{het}.
	\item Añadir artículo donde no va con profesiones: \textit{Hij is leraar}, no \textit{een leraar} en definiciones.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Clasifica 30 palabras en \textit{de/het} y comprueba con respuesta.
	\item Completa frases eligiendo \textit{de/het/een} según contexto.
	\item Transforma 8 frases singulares a plural ajustando el artículo.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Describe 6 objetos de tu habitación usando \textit{de/het/een} correctamente.
	\item Graba 8 frases combinando artículo + adjetivo + sustantivo (\textit{een grote tafel}, \textit{het kleine huis}).
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Reconoces los het-woorden más frecuentes de A1.
	\item Ajustas artículo en plural automáticamente.
	\item Reduces \textit{het/een} en habla rápida sin perder claridad.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes elegir el artículo correcto en frases cotidianas y sonar natural al reducir \textit{het/een} en conversación.
\end{itemize}
