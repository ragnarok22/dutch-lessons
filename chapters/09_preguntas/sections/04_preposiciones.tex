\section{Preguntas con preposición}
\textbf{Patrón:} Preposición + wh-word o al final (estilo coloquial aceptado, pero mejor mantenerla con la wh cuando sea posible).

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Waar woon je?} (dónde)
  \item \textit{Waarmee kom je?} (con qué)
  \item \textit{Met wie ga je?} (con quién)
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Forma 6 preguntas con preposiciones \textit{met, naar, bij, over, voor, van}.
  \item Responde en una frase completa para fijar orden.
\end{itemize}
