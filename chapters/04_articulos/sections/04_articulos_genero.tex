\section{Artículos y género}
\textbf{Qué es:} Relación artículo-género y estrategias para recordar.

\textbf{Estrategias prácticas}
\begin{itemize}[leftmargin=*]
  \item Aprende el artículo con la palabra desde el inicio (flashcards: cara NL con \textit{de/het}).
  \item Marca patrones de het-woorden (diminutivos, idiomas, materiales) y asume \textit{de} como default.
  \item Usa colores o tipografías distintas en tus apuntes para de/het.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Reescribe tu glosario de la semana añadiendo artículo correcto.
  \item Juego de clasificación: 20 palabras mezcladas, cronómetro a 2 minutos.
\end{itemize}
