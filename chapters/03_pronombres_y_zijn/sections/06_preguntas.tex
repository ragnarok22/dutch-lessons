\section{Preguntas con \textit{zijn}}
\textbf{Sí/no:} Verbo en primera posición.
\begin{itemize}[leftmargin=*]
  \item \textit{Ben je klaar?} \textit{Is hij thuis?}
\end{itemize}

\textbf{Wh- preguntas:} Palabra interrogativa + verbo + sujeto.
\begin{itemize}[leftmargin=*]
  \item \textit{Waar ben je?} \textit{Wie is dat?} \textit{Hoe zijn jullie?}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 6 preguntas (3 sí/no, 3 wh-). Responde en dos versiones: corta y completa.
  \item Role-play: entrevista rápida de 6 turnos usando \textit{zijn}.
\end{itemize}
