\section{Profesiones y trabajos}
\textbf{Vocabulario inicial.}
\begin{itemize}[leftmargin=*]
  \item Profesiones comunes: \textit{docent, verpleegkundige, ingenieur, verkoper, student, kok, chauffeur, arts}.
  \item Lugares asociados: \textit{school, ziekenhuis, kantoor, winkel, restaurant, bus}.
\end{itemize}

\textbf{Frases modelo}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik ben leraar. Ik werk op een school}.
  \item \textit{Zij is verpleegkundige in een ziekenhuis}.
  \item \textit{Werk je fulltime of parttime?}
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe dos frases sobre tu profesión o estudios.
  \item Pregunta a un compañero dónde trabaja y cuántas horas.
\end{itemize}
