\section{Infinitivo y raíz del verbo}
\textbf{Objetivo:} Identificar la raíz quitando \textit{-en} del infinitivo y ajustar ortografía básica.

\textbf{Pasos}
\begin{itemize}[leftmargin=*]
  \item Infinitivo termina en \textit{-en}: \textit{werken, wonen, leren, spelen}.
  \item Raíz = infinitivo sin \textit{-en}: \textit{werk, woon, leer, speel}.
  \item Ajustes: si la raíz termina en doble consonante o vocal larga, cuida no perder la longitud al añadir terminaciones.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Extrae raíz de 10 verbos comunes y léela en voz alta.
  \item Observa cambios: \textit{leven} \textrightarrow{} \textit{leef-}, \textit{reizen} \textrightarrow{} \textit{reis-}.
\end{itemize}
