\chapter{Rutina diaria y trabajo}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Describir rutinas diarias y horarios laborales simples.
	\item Usar adverbios de frecuencia y conectores básicos.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Verbos de rutina: levantarse, comer, trabajar, estudiar, volver.
	\item Adverbios: \textit{altijd, meestal, soms, nooit, elke dag, \textquotesingle{}s morgens}.
	\item Conectores: \textit{en, maar, want, daarna, dan} (orden V2 con conectores iniciales).
\end{itemize}
