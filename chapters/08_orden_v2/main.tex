\chapter{Orden de la frase (V2)}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
  \item Aplicar la regla de verbo en segunda posición en enunciados afirmativos.
  \item Mover el foco inicial (tiempo/lugar) manteniendo V2.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
  \item Estructura básica: [Sujeto/Elemento inicial] + Verbo + Complementos.
  \item Inversión sujeto-verbo al anteponer adverbios de tiempo/lugar: \textit{Morgen ga ik naar school}.
  \item Posición de negación básica y partículas verbales separables (vista previa).
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
  \item Reordenar frases para practicar V2.
  \item Combinar con expresiones temporales y adverbios iniciales.
  \item Mini narraciones en tres frases cambiando el elemento inicial.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
  \item 12 frases: 6 con sujeto inicial, 6 con tiempo/lugar inicial, todas con V2 correcto.
\end{itemize}
