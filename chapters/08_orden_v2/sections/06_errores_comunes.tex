\section{Errores comunes de orden}
\textbf{Qué evitar:}
\begin{itemize}[leftmargin=*]
  \item Verbo en tercera posición al mover adverbios: *\textit{Morgen ik ga}. Corrige: \textit{Morgen ga ik}.
  \item Olvidar invertir sujeto-verbo tras elemento inicial distinto del sujeto.
  \item Colocar \textit{niet} antes del verbo principal en frases simples (se pone después del resto).
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Corrige 8 frases con errores típicos de orden.
  \item Lee en voz alta la versión corregida para fijar el patrón.
\end{itemize}
