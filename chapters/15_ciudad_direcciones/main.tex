\chapter{La ciudad y direcciones}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Comprender y dar indicaciones básicas en la ciudad.
	\item Manejar vocabulario de transporte y lugares públicos.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Lugares: estación, parada, supermercado, oficina de correos, plaza, calle.
	\item Direcciones: \textit{rechtdoor, linksaf, rechtsaf, tegenover, naast, vlakbij, ver}.
	\item Preguntas modelo: \textit{Waar is \ldots{}? Hoe kom ik bij \ldots{}? Is het ver?}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Acentúa la primera sílaba en \textit{rechtdoor} /ˈrɛxt.doːr/ y \textit{linksaf} /ˈlɪŋks.ɑf/.
	\item Mantén /x/ suave en \textit{rechtsaf}; evita /g/ español.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Usar preposición incorrecta: \textit{naast} (al lado), \textit{tegenover} (enfrente), \textit{bij} (junto).
	\item Olvidar V2 en instrucciones: \textit{Ga rechtdoor}, no \textit{Jij ga rechtdoor}.
	\item Confundir \textit{links} (a la izquierda) con \textit{linker-} (lado izquierdo de algo).
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Sigue rutas en un mapa simple y completa huecos con preposiciones.
	\item Ordena frases de dirección con V2 correcto.
	\item Tarjetas de señales: identifica y formula instrucciones asociadas.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Describe tu ruta diaria al trabajo/estudio en 8 frases.
	\item Graba dos diálogos pidiendo y dando indicaciones con confirmaciones (\textit{Dus ik ga\ldots{}}).
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Preguntas y entiendes indicaciones básicas en ciudad.
	\item Usas preposiciones de lugar correctas y V2 en imperativos.
	\item Reconoces señales comunes y las explicas brevemente.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes pedir y dar direcciones simples, interpretar señalética básica y describir rutas cortas con vocabulario de transporte.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Mapas sencillos: marcar rutas y describirlas en V2.
	\item Diálogo de pedir indicaciones y confirmación.
	\item Señalética: leer y escribir carteles cortos.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
