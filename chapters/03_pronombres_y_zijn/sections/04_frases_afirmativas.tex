\section{Frases afirmativas con \textit{zijn}}
\textbf{Estructura básica:} Sujeto + forma de \textit{zijn} + atributo/complemento.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik ben leraar.}
  \item \textit{Jij bent Nederlands.}
  \item \textit{We zijn blij.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Sustituye sujeto y atributo: 8 combinaciones distintas (personas × profesiones/estados).
  \item Lee en cadena: cada frase con diferente persona gramatical.
\end{itemize}
