\section{Medios de transporte}
\textbf{Vocabulario.}
\begin{itemize}[leftmargin=*]
  \item Vehículos: \textit{auto/auto}, \textit{fiets}, \textit{bus}, \textit{trein}, \textit{tram}, \textit{metro}, \textit{taxi}.
  \item Billetes: \textit{een kaartje}, \textit{retour}, \textit{enkele reis} (solo ida), \textit{abonnement}.
  \item Acciones: \textit{instappen/uitstappen} (subir/bajar), \textit{overstappen} (hacer transbordo).
\end{itemize}

\textbf{Frases modelo}
\begin{itemize}[leftmargin=*]
  \item \textit{Waar kan ik een kaartje kopen?}
  \item \textit{Ik stap uit bij Centraal Station}.
  \item \textit{We moeten overstappen in Utrecht}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Describe tu trayecto diario usando dos medios de transporte.
  \item Haz dos preguntas sobre billetes o abonos.
\end{itemize}
