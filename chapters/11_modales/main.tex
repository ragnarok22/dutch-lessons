\chapter{Verbos modales básicos}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Usar modales frecuentes en presente para expresar capacidad, permiso y deseo.
	\item Practicar infinitivos al final de la oración principal.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Modales: \textit{kunnen, mogen, willen, moeten, zullen} (uso A1 de futuro/ intención con \textit{zullen}).
	\item Orden: [Sujeto] + [Modal conjugado] + \ldots{} + [Infinitivo final].
	\item Fórmulas de cortesía con \textit{mogen/kunnen}.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Verbos de apoyo: \textit{werken}, \textit{gaan}, \textit{koken}, \textit{helpen}, \textit{spreken}, \textit{kopen}.
	\item Marcadores de cortesía: \textit{alsjeblieft/alstublieft}, \textit{mag ik}, \textit{kan ik}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Mantén la /x/ suave en \textit{mag} y evita /g/ duro.
	\item \textit{wil} suena /ʋɪl/; no alargues la vocal como en español.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Olvidar el infinitivo al final: \textit{Ik kan niet komen}, no \textit{Ik kan niet}.
	\item Repetir modal: evita \textit{Ik wil kan...}; usa solo uno.
	\item Usar \textit{moeten} cuando se necesita permiso (\textit{mogen}).
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Completa frases con modal correcto según intención (capacidad, permiso, deseo, obligación).
	\item Reordena 10 frases colocando el infinitivo al final.
	\item Role-play guiado de peticiones y permisos en tienda/cafetería.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe 8 frases de obligación/deseo/plan (mezcla modales).
	\item Graba 6 peticiones educadas y 6 respuestas afirmativas/negativas.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Seleccionas el modal correcto según intención.
	\item Colocas el infinitivo al final de la oración principal.
	\item Usas fórmulas de cortesía con buena pronunciación.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes pedir, ofrecer, obligar o expresar deseo con los cinco modales básicos manteniendo el orden holandés.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Completar oraciones con modal + infinitivo.
	\item Role-play de peticiones y permisos en tienda/cafetería.
	\item Transformar deseos en planes con \textit{zullen}.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item 10 frases grabadas mezclando los cinco modales con verbos conocidos.
\end{itemize}
