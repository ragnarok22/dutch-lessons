\section{Sustantivos con \textit{het}}
\textbf{Qué es:} Artículo definido para género neutro (het-woorden). Menos frecuente, pero con patrones útiles.

\textbf{Patrones comunes}
\begin{itemize}[leftmargin=*]
  \item Diminutivos en \textit{-je}: \textit{het huisje, het boekje}.
  \item Idiomas y materiales: \textit{het Nederlands, het hout}.
  \item Palabras prestadas recientes y muchos nombres de países: \textit{het museum, het Spanje} (variable con países; se aprenden caso a caso).
\end{itemize}
