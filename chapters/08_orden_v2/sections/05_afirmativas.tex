\section{Orden en frases afirmativas}
\textbf{Objetivo:} Consolidar V2 en enunciados positivos con complementos.

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Vandaag eet hij pasta thuis.}
  \item \textit{Na het werk kijk ik tv.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Completa 10 oraciones con bloques dados (tiempo + verbo + sujeto + lugar/objeto).
  \item Combina con partículas separables sencillas: \textit{Ik sta om zeven uur op. Morgen sta ik om zes uur op}.
\end{itemize}
