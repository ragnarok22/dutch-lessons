\chapter{Sustantivos y plural}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Formar plurales regulares e identificar plurales irregulares frecuentes.
	\item Combinar plural con numerales y determinantes básicos.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Reglas de plural en \textit{-en} y \textit{-s}.
	\item Cambios ortográficos frecuentes: duplicación y caída de vocal/consonante.
	\item Plurales irregulares esenciales A1.
	\item Uso del plural con números, \textit{geen}, demostrativos y posesivos.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Ejemplos regulares: \textit{tafel/tafels}, \textit{auto/auto's}, \textit{boek/boeken}.
	\item Irregulares frecuentes: \textit{kind/kinderen}, \textit{stad/steden}, \textit{blad/bladeren}.
	\item Contables básicos: \textit{stuk}, \textit{fles}, \textit{kop}, \textit{zak}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Suaviza la /en/ final en plurales en \textit{-en}. Ejemplo: \textit{boeken} → \textipa{/bu.k@n/}.
	\item Marca apóstrofo en \textit{auto's} con ligera pausa, sin añadir sílaba extra.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Añadir \textit{-s} a palabras que requieren \textit{-en} (\\textit{tafel} → \textit{tafels} ok; \\textit{boek} no \textit{boeks} en A1).
	\item Olvidar cambios ortográficos: \\textit{stad} → \textit{steden}, no \textit{stads}.
	\item Repetir artículo singular con número: usa solo número + sustantivo plural (\textit{twee boeken}).
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Completa tablas singular→plural con corrección inmediata.
	\item Elige artículo/determinante correcto con números y \textit{geen}.
	\item Transforma frases singulares a plural ajustando verbo ser/estar si aplica.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe 10 frases con cantidades y plurales de tu entorno (cocina/oficina).
	\item Graba 8 frases usando plurales irregulares y regulares mezclados.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Aplicas reglas \textit{-en}/\textit{-s} y cambios ortográficos frecuentes.
	\item No usas artículo con números en plural.
	\item Reconoces y produces los irregulares clave de A1.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes formar y usar plurales comunes con números y determinantes sin dudar en conversación básica.
\end{itemize}
