\chapter{Rutina diaria y trabajo}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Describir rutinas diarias y horarios laborales simples.
	\item Usar adverbios de frecuencia y conectores básicos.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Verbos de rutina: levantarse, comer, trabajar, estudiar, volver.
	\item Adverbios: \textit{altijd, meestal, soms, nooit, elke dag, \textquotesingle{}s morgens}.
	\item Conectores: \textit{en, maar, want, daarna, dan} (orden V2 con conectores iniciales).
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Reduce la \textit{-en} final en verbos de rutina: \textit{werken} /ˈʋɛrkə(n)/.
	\item Marca /x/ suave en \textit{acht uur}, \textit{acht-en-dertig}.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Olvidar V2 tras adverbio inicial (\textit{Daarna ga ik werken}).
	\item Confundir \textit{altijd} con \textit{elke dag}; matiza frecuencia.
	\item Usar presente progresivo inglés; en neerlandés usa presente simple.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Línea de tiempo: coloca verbos y adverbios en orden con V2 correcto.
	\item Completa cuadros de horarios y conecta con \textit{daarna/dan}.
	\item Rellena diálogos sobre profesión y lugar de trabajo.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Escribe tu rutina diaria en 10 frases con horarios.
	\item Graba 8 frases sobre tu trabajo/estudios usando adverbios de frecuencia.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Relatas rutina con V2 y conectores básicos.
	\item Colocas adverbios de frecuencia en posición natural.
	\item Nombras profesión y lugar de trabajo con vocabulario A1.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes describir tu día, tus horarios y tu trabajo/estudios de forma clara y ordenada en A1.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
	\item Redacción corta de rutina con horarios y frecuencia.
	\item Entrevista cruzada sobre profesiones y horarios.
	\item Línea de tiempo visual y narración oral.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Texto de 120 palabras o audio de 1 minuto describiendo tu día típico.
\end{itemize}
