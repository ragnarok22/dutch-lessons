\section{Actividades diarias}
\textbf{Combinaciones útiles.}
\begin{itemize}[leftmargin=*]
  \item Transporte: \textit{met de fiets/bus/auto gaan}.
  \item Compras y recados: \textit{boodschappen doen}, \textit{naar de supermarkt gaan}.
  \item Ocio breve: \textit{tv kijken}, \textit{wandelen}, \textit{lezen}.
\end{itemize}

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik ga met de fiets naar het werk}.
  \item \textit{Na het werk doe ik boodschappen}.
  \item \textit{We lezen \'s avonds een boek}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe tres frases con transporte + destino.
  \item Combina una acción laboral y una de ocio en una sola oración.
\end{itemize}
