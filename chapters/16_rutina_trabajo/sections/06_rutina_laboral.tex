\section{Rutina laboral básica}
\textbf{Estructura de un día de trabajo.}
\begin{itemize}[leftmargin=*]
  \item Inicio: \textit{Ik begin om \\underline{\hspace{1.2cm}}}. 
  \item Tareas: \textit{Ik lees e-mails}, \textit{Ik heb een meeting}, \textit{Ik bel klanten}, \textit{Ik schrijf rapporten}.
  \item Pausas: \textit{Ik neem pauze om twaalf uur}. 
  \item Cierre: \textit{Ik stop om vijf uur en ga naar huis}. 
\end{itemize}

\textbf{Modelo corto}
\begin{itemize}[leftmargin=*]
  \item \textit{Ik begin om acht uur, ik werk tot vijf uur, en daarna kook ik}. 
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe tu rutina laboral en 4 líneas siguiendo el modelo.
  \item Graba o lee en voz alta tu texto.
\end{itemize}
