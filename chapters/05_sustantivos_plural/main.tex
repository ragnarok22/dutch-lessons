\chapter{Sustantivos y plural}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
  \item Formar plurales regulares e identificar plurales irregulares frecuentes.
  \item Combinar plural con numerales y determinantes básicos.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
  \item Reglas: sufijos \textit{-en}, \textit{-s}; cambios vocálicos básicos (\textit{stad/steden}).
  \item Diminutivos \textit{-je} y su plural \textit{-tjes}.
  \item Uso con números: sin artículo, singular para 1, plural 2+.
\end{itemize}

\section{Práctica sugerida}
\begin{itemize}[leftmargin=*]
  \item Convertir listas de singular a plural y viceversa.
  \item Completar diálogos con cantidades y plurales.
  \item Dictado de palabras y pluralización oral.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
  \item Cuadro de 15 sustantivos: escribe plural y úsalo en una frase.
\end{itemize}
