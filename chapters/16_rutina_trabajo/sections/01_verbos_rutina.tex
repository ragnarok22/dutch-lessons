\section{Verbos de rutina diaria}
\textbf{Lista base.}
\begin{itemize}[leftmargin=*]
  \item \textit{opstaan} (levantarse), \textit{douchen}, \textit{aankleden} (vestirse), \textit{ontbijten}, \textit{werken}, \textit{studeren}, \textit{koken}, \textit{slapen}.
  \item Verbos separables frecuentes: \textit{opstaan}, \textit{thuiskomen} (volver a casa), \textit{meebrengen} (traer), \textit{uitgaan} (salir de ocio).
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Conjuga cinco verbos de la lista para \textit{ik} y \textit{wij}.
  \item Escribe 4 frases cortas describiendo acciones diarias.
\end{itemize}
