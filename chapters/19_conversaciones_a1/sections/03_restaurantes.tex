\section{Conversaciones en restaurantes}
\textbf{Diálogo básico.}
\begin{itemize}[leftmargin=*]
  \item Mesa: \textit{Heeft u een tafel voor twee?}
  \item Pedido: \textit{Ik wil graag de soep en water}.
  \item Preferencias: \textit{Heeft u iets vegetarisch? Zonder kaas, alstublieft}.
  \item Cuenta: \textit{Mag ik de rekening, alstublieft?}
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe un pedido completo (bebida + plato) y una pregunta al camarero.
  \item Role-play breve cliente/camarero.
\end{itemize}
