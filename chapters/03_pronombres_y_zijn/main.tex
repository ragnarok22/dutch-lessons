\chapter{Pronombres personales y verbo \textit{zijn}}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Memorizar los pronombres sujeto y las formas del verbo \textit{zijn} en presente.
	\item Construir oraciones básicas afirmativas, interrogativas y negativas con \textit{zijn}.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Pronombres sujeto: \textit{ik, jij/je, u, hij/zij/het, wij, jullie, zij}.
	\item Conjugación completa de \textit{zijn} y formas invertidas para preguntas.
	\item Orden de palabras en afirmaciones, negaciones y preguntas sí/no y \textit{wh}.
	\item Nacionalidad y profesión con \textit{zijn} + adjetivo o sustantivo.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Nacionalidades: \textit{Nederlands}, \textit{Belgisch}, \textit{Spaans}, \textit{Mexicaans}.
	\item Profesiones básicas: \textit{student}, \textit{leraar}, \textit{arts}, \textit{ingenieur}, \textit{verkoper}.
	\item Pronombres de cortesía: \textit{u}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Reduce la \textit{n} final en \textit{zijn} en habla rápida: /zɛi̯n/.
	\item Marca la /x/ suave en \textit{Belgisch}; evita \textit{bél-jis}.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Omitir inversión en preguntas: usa \textit{Ben je ... ? / Is hij ... ?}.
	\item Confundir \textit{u} con \textit{jij}; mantén la formalidad con desconocidos.
	\item Colocar \textit{niet} al inicio; en negaciones simples va al final.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Completa una tabla de pronombres + forma de \textit{zijn}.
	\item Transforma 8 frases afirmativas en negativas y luego en preguntas.
	\item Rellena diálogos con nacionalidad y profesión.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Graba una presentación de 8 frases usando todas las personas.
	\item Escribe 6 preguntas y 6 respuestas sobre identidad y profesión con \textit{zijn}.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Conjugas \textit{zijn} para todas las personas.
	\item Formas preguntas invertidas con fluidez.
	\item Expresas nacionalidad y profesión sin artículo.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes presentarte, decir qué haces y formular/contestar preguntas con \textit{zijn} usando el orden correcto.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Escribe y lee en voz alta 10 frases usando todas las personas gramaticales.
\end{itemize}
