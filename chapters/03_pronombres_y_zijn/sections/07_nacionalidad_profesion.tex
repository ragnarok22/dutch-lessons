\section{Nacionalidad y profesión con \textit{zijn}}
\textbf{Objetivo:} Decir quién eres y de dónde eres en términos de nacionalidad/profesión.

\textbf{Estructuras}
\begin{itemize}[leftmargin=*]
  \item Nacionalidad: \textit{Ik ben Mexicaans. Zij is Nederlands.} (adjetivo demonymio con \textit{-s} en muchos casos).
  \item Profesión: \textit{Ik ben ingenieur. Hij is kok.}
  \item Combinado: \textit{Ik ben Spaanse lerares.}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Crea 8 frases mezclando nacionalidad y profesión.
  \item Intercambia en parejas: pregunta/responde \textit{Waar kom je vandaan? Wat ben je van beroep?}
\end{itemize}
