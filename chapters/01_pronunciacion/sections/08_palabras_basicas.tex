\section{Pronunciación de palabras básicas}
\textbf{Qué es:} Lista corta de alta frecuencia para consolidar sonidos clave.

\textbf{Cómo usarla:}
\begin{itemize}[leftmargin=*]
  \item Lee y graba pares con distintos sonidos: \textit{huis, deur, tijd, vrouw, school, rood, werken}.
  \item Haz tarjetas con audio (TTS NL) y repite hasta que tu versión se acerque.
\end{itemize}

\textbf{Mini lista sugerida:}
\begin{itemize}[leftmargin=*]
  \item \textit{hallo, goed, huis, deur, tijd, auto, fietsen, straat, groen, leuk, werken, leren}.
\end{itemize}
