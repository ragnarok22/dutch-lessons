\section{Plurales irregulares comunes}
\textbf{Qué son:} Formas que no siguen \textit{-en/-s} directas.

\textbf{Lista esencial A1}
\begin{itemize}[leftmargin=*]
  \item \textit{stad \textrightarrow{} steden}
  \item \textit{kind \textrightarrow{} kinderen}
  \item \textit{ei \textrightarrow{} eieren}
  \item \textit{blad \textrightarrow{} bladeren}
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Memoriza con tarjetas (singular delante, plural detrás).
  \item Escribe 6 frases usando cada irregular en contexto.
\end{itemize}
