\section{Adjetivos comunes}
\textbf{Vocabulario.}
\begin{itemize}[leftmargin=*]
  \item Tamaño/forma: \textit{groot, klein, lang, kort, rond}. 
  \item Estado: \textit{nieuw, oud, schoon, vuil, kapot}. 
  \item Carácter: \textit{aardig, rustig, druk, grappig}. 
\end{itemize}

\textbf{Ejemplos}
\begin{itemize}[leftmargin=*]
  \item \textit{Mijn fiets is oud maar sterk}. 
  \item \textit{Zij is rustig}. 
  \item \textit{Het huis is groot en nieuw}. 
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Escribe 5 frases cortas con adjetivos de tamaño/estado.
  \item Describe a una persona con dos adjetivos de carácter.
\end{itemize}
