\section{Lugares de la ciudad}
\textbf{Vocabulario esencial.}
\begin{itemize}[leftmargin=*]
  \item Transporte: \textit{station}, \textit{bushalte}, \textit{trein, bus, tram, metro}.
  \item Servicios: \textit{supermarkt}, \textit{apotheek}, \textit{postkantoor}, \textit{bank}, \textit{politiebureau}.
  \item Ocio y espacios: \textit{park}, \textit{museum}, \textit{plein}, \textit{bibliotheek}, \textit{zwembad}.
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Marca en un mapa tres lugares y nómbralos en neerlandés.
  \item Escribe dos frases con \textit{er is/er zijn} para describir tu barrio.
\end{itemize}
