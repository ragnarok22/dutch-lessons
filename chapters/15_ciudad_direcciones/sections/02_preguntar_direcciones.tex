\section{Preguntar direcciones}
\textbf{Frases clave.}
\begin{itemize}[leftmargin=*]
  \item \textit{Waar is \\_?} (¿Dónde está \_?)
  \item \textit{Hoe kom ik bij \\_?} (¿Cómo llego a \_?)
  \item \textit{Is het ver?} (¿Está lejos?) / \textit{Hoe lang duurt het?}
  \item \textit{Kunt u dat herhalen, alstublieft?}
\end{itemize}

\textbf{Mini práctica}
\begin{itemize}[leftmargin=*]
  \item Haz tres preguntas distintas para llegar a un lugar.
  \item Role-play turista/ciudadano: pide que repitan la indicación.
\end{itemize}
