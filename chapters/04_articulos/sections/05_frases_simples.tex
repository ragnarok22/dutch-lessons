\section{Artículos en frases simples}
\textbf{Objetivo:} Insertar de/het/een correctamente en oraciones básicas.

\textbf{Plantillas}
\begin{itemize}[leftmargin=*]
  \item \textit{Dit is \ldots{}} (señalar objeto): \textit{Dit is de/het \ldots{}}.
  \item \textit{Ik heb \ldots{}} + sustantivo: \textit{Ik heb een fiets. Ik heb de sleutel.}
  \item Locaciones: \textit{De sleutel is op de tafel}.~\textit{Het boek is in het huis}.
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Completa 12 oraciones huecas con el artículo correcto.
  \item Lee en voz alta cuidando la reducción de \textit{een} y el enlace con preposiciones: \textit{op de}, \textit{in het}.
\end{itemize}
