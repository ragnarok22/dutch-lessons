\section{Uso de \textit{een}}
\textbf{Qué es:} Artículo indefinido (``un/una'') sin énfasis de acento, se pronuncia casi sin vocal plena.

\textbf{Cuándo usarlo}
\begin{itemize}[leftmargin=*]
  \item Sustantivo singular no específico: \textit{een boek, een vrouw}.
  \item Presentaciones y descripciones generales: \textit{Ik ben een student}. 
\end{itemize}

\textbf{Cuándo no usarlo}
\begin{itemize}[leftmargin=*]
\item Plural: se omite artículo (\textit{boeken}).
  \item Sustantivos incontables o genéricos: \textit{Ik drink water}. 
\end{itemize}

\textbf{Práctica}
\begin{itemize}[leftmargin=*]
  \item Completa 10 frases con \textit{een} o sin artículo.
  \item Escucha y repite fijándote en la reducción fonética de \textit{een} (\textipa{/@n/}, sonido schwa muy breve).
\end{itemize}
