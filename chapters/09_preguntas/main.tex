\chapter{Preguntas y palabras interrogativas}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Formular preguntas de sí/no y de información respetando la posición del verbo.
	\item Manejar palabras interrogativas básicas.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Inversión verbo-sujeto en preguntas sí/no.
	\item Palabras interrogativas: \textit{wie, wat, waar, wanneer, waarom, hoe}.
	\item Preguntas con preposición al final y respuestas cortas.
	\item Entonación ascendente y marcas de cortesía.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Interrogativos: \textit{wie, wat, waar, wanneer, waarom, hoe, hoeveel}.
	\item Preposiciones comunes en preguntas: \textit{met, voor, bij, naar, van}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Eleva entonación al final en sí/no; cae ligeramente en wh- preguntas.
	\item Mantén \textipa{/v/} suave en \textit{waar} y \textit{wanneer}.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Omitir inversión: evita \textit{Je gaat waar?}; usa \textit{Waar ga je?}
	\item Colocar preposición al inicio por calco del español; déjala al final: \textit{Waar kom je vandaan?}
	\item Repetir sujeto y pronombre: usa uno solo tras el verbo.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Reescribe 12 afirmaciones en preguntas sí/no con inversión.
	\item Completa preguntas con wh- adecuada según respuesta dada.
	\item Une respuesta corta correcta a cada pregunta.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Graba 10 preguntas (5 sí/no, 5 wh) y contesta de forma natural.
	\item Crea un mini cuestionario de 8 preguntas para conocer a un compañero.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Inviertes verbo-sujeto sin dudar en preguntas.
	\item Sitúas la preposición al final cuando corresponde.
	\item Ajustas entonación según tipo de pregunta.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes hacer y responder preguntas básicas con inversión correcta, wh- apropiadas y entonación natural.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item 10 preguntas orales grabadas + respuestas cortas.
\end{itemize}
