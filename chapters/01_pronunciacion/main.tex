\chapter{Pronunciación y alfabeto}
\section{Objetivos}
\begin{itemize}[leftmargin=*]
	\item Reconocer el alfabeto neerlandés y sus sonidos principales.
	\item Producir vocales largas/cortas y diptongos con buena aproximación.
	\item Diferenciar consonantes problemáticas: \textit{g/ch}, \textit{r}, \textit{ui}, \textit{ij/ei}.
\end{itemize}

\section{Contenidos clave}
\begin{itemize}[leftmargin=*]
	\item Alfabeto y nombres de letras.
	\item Vocales cortas vs. largas y diptongos frecuentes.
	\item Consonantes fricativas y \textit{r} en posición inicial/final.
	\item Acento de palabra y sílabas abiertas/cerradas.
\end{itemize}

\section{Vocabulario clave}
\begin{itemize}[leftmargin=*]
	\item Letras con sonido distinto al español: \textit{g}, \textit{ch}, \textit{ui}, \textit{ij/ei}, \textit{oo}, \textit{uu}.
	\item Palabras ancla para practicar: \textit{goed}, \textit{huis}, \textit{werken}, \textit{rijden}, \textit{vriend}.
\end{itemize}

\section{Pronunciación}
\begin{itemize}[leftmargin=*]
	\item Mantén la \textit{g/ch} velar sin vocalizar; evita suavizarla como en español.
	\item Alarga las vocales dobles al menos el doble de tiempo que las cortas.
	\item Marca el acento en la primera sílaba salvo préstamos; escucha y repite.
\end{itemize}

\section{Errores comunes}
\begin{itemize}[leftmargin=*]
	\item Confundir \textit{ui} con \textit{oe}; practica pares mínimos \textit{huis} vs. \textit{hoes}.
	\item Reducir \textit{r} final a vocal; mantén un toque breve o uvular.
	\item Perder el acento de palabra al leer lento; añade palmas para ritmo.
\end{itemize}

\section{Práctica guiada}
\begin{itemize}[leftmargin=*]
	\item Lee el alfabeto grabándote y compara con el audio modelo.
	\item Repite cadenas consonante-vocal: \textit{ka, ke, ki, ko, ku} y \textit{ga, ge, gi, go, gu}.
	\item Pares mínimos de diptongos: \textit{huis/hoed}, \textit{rijden/reiden}.
\end{itemize}

\section{Ejercicios de producción}
\begin{itemize}[leftmargin=*]
	\item Graba 10 palabras con \textit{g/ch/ui/ij} y una frase corta con cada una.
	\item Lee en voz alta un párrafo breve (20--30 palabras) marcando acento.
\end{itemize}

\section{Mini repaso}
\begin{itemize}[leftmargin=*]
	\item Diferencia vocales cortas vs. largas y diptongos clave.
	\item Coloca acento correcto en palabras comunes.
	\item Produces \textit{g/ch} sin suavizarlas.
	\item Repites pares mínimos con buena precisión.
\end{itemize}

\section{Resultado esperado}
\begin{itemize}[leftmargin=*]
	\item Al terminar puedes leer el alfabeto, pronunciar vocales y consonantes difíciles, y grabar frases básicas con ritmo natural.
\end{itemize}

\section{Microevaluación}
\begin{itemize}[leftmargin=*]
	\item Graba un minuto presentando el alfabeto y tu nombre; compáralo con un modelo nativo.
\end{itemize}
